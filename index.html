// 1. تحميل البيانات وبناء الشبكة والقائمة المنسدلة fetch('./sunan_data.json') .then(response => response.json()) .then(data => { // إنشاء مجموعات البيانات لـ Vis Network const nodes = new vis.DataSet(data.nodes); const edges = new vis.DataSet(data.edges); // إعداد حاوي الشبكة والخيارات const container = document.getElementById('network-container'); const options = { physics: { enabled: true }, nodes: { shape: 'dot', size: 16 } }; // تشغيل الشبكة const network = new vis.Network(container, { nodes, edges }, options); // 2. تفعيل القائمة المنسدلة للسنن الـ 95 (Dropdown Selector) const selector = document.getElementById('sunan-select'); // تأكد من id القائمة لديك if (selector) { // تعبئة القائمة تلقائياً data.nodes.forEach(node => { const opt = document.createElement('option'); opt.value = node.id; opt.textContent = `${node.id}. ${node.label}`; selector.appendChild(opt); }); // الاستجابة عند اختيار سنة معينة selector.addEventListener('change', (e) => { const selectedId = e.target.value; if (selectedId) { // التركيز على العُقدة المختارة والتكبير عليها network.focus(selectedId, { scale: 1.2, animation: { duration: 1000, easingFunction: 'easeInOutQuad' } }); network.selectNodes([selectedId]); // إظهار التفاصيل في اللوحة الجانبية const selectedNode = nodes.get(selectedId); showSidebarDetails(selectedNode); } }); } // 3. الاستجابة عند النقر المباشر على أي عقدة داخل الشبكة network.on("click", function (params) { if (params.nodes.length > 0) { const nodeId = params.nodes[0]; const selectedNode = nodes.get(nodeId); if (selector) selector.value = nodeId; // تحديث القائمة المنسدلة showSidebarDetails(selectedNode); } }); }) .catch(err => console.error('خطأ في تحميل البيانات:', err)); // دالة عرض التفاصيل في اللوحة الجانبية function showSidebarDetails(node) { const sidebar = document.getElementById('sidebar-content'); if (sidebar && node) { sidebar.innerHTML = `

${node.label}

المجال: ${node.group || node.domain || 'عام'}

المعادلة السببية: ${node.formula || 'غير محددة'}

المقاصد: ${node.details || ''}

`; } }
